import logging
import sys


def setup_logger(name: str = "rpg") -> logging.Logger:
    """Единый логгер. Пишет в stdout — Docker и systemd журналируют его сами."""
    logger = logging.getLogger(name)
    if logger.handlers:  # защита от дублирования при повторном импорте
        return logger
    logger.setLevel(logging.INFO)
    handler = logging.StreamHandler(sys.stdout)
    handler.setFormatter(logging.Formatter(
        "%(asctime)s | %(levelname)-8s | %(name)s | %(message)s"
    ))
    logger.addHandler(handler)
    return logger


logger = setup_logger()

from functools import lru_cache
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    """Все настройки — из .env. Ни одного секрета в коде!"""

    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore",  # не падать, если в окружении есть лишние переменные
    )

    # OpenAI
    openai_api_key: str
    model: str = "gpt-4o-mini"      # дёшево. Для продакшена: "gpt-4o"
    max_history_length: int = 20
    openai_timeout: float = 60.0

    # Rate limiting: N запросов за окно (сек) на одного пользователя
    rate_limit_requests: int = 10
    rate_limit_window: int = 60

    system_prompt: str = (
        "Ты — опытный мастер Dungeons & Dragons 5e. "
        "Описывай сцены красочно, но лаконично (до 200 слов). "
        "После каждого описания предлагай игроку 2-3 варианта действий списком. "
        "Используй Markdown: **жирный** для важного, списки для вариантов. "
        "Если игрок сообщает результат броска кубика — учитывай его по правилам D&D 5e. "
        "Никогда не выходи из роли мастера игры."
    )

    app_name: str = "D&D AI Master"
    debug: bool = False
    cors_origins: list[str] = ["*"]


@lru_cache()
def get_settings() -> Settings:
    return Settings()

from pydantic import BaseModel, Field, field_validator


class ChatRequest(BaseModel):
    user_id: str = Field(..., min_length=1, max_length=64,
                         description="Уникальный ID пользователя (UUID)")
    message: str = Field(..., min_length=1, max_length=4000,
                         description="Сообщение игрока")

    @field_validator("user_id", "message")
    @classmethod
    def not_blank(cls, v: str) -> str:
        if not v.strip():
            raise ValueError("Поле не может состоять из пробелов")
        return v.strip()


class ChatResponse(BaseModel):
    reply: str = Field(..., description="Ответ AI-мастера (Markdown)")
    timestamp: str = Field(..., description="Время ответа, ISO 8601 UTC")


class HealthResponse(BaseModel):
    status: str
    version: str = "1.0.0"

2.7. app/services/rate_limiter.py

import time
from collections import defaultdict, deque
from app.config import settings
from app.exceptions import RateLimitError


class InMemoryRateLimiter:
    """
    Скользящее окно: хранит время последних запросов каждого пользователя.
    Для MVP достаточно; с несколькими воркерами нужен Redis (Этап 13).
    """

    def __init__(self):
        self._requests: dict[str, deque] = defaultdict(deque)

    def check(self, user_id: str) -> None:
        now = time.monotonic()
        window = settings.rate_limit_window
        q = self._requests[user_id]

        while q and now - q[0] > window:  # выбрасываем устаревшие записи
            q.popleft()

        if len(q) >= settings.rate_limit_requests:
            raise RateLimitError(
                f"Слишком много запросов. Подождите {window} секунд."
            )
        q.append(now)


rate_limiter = InMemoryRateLimiter()

2.8. app/services/chat_service.py

from collections import defaultdict
from datetime import datetime, timezone
from typing import Dict, List
from openai import AsyncOpenAI, APIError, RateLimitError as OpenAIRateLimit
from app.config import settings
from app.models import ChatRequest, ChatResponse
from app.exceptions import OpenAIError, RateLimitError
from app.logger import logger


class ChatService:
    """
    История диалогов + общение с OpenAI.
    ⚠️ История в памяти: теряется при рестарте и НЕ работает с >1 воркером.
    Для MVP — норма. Для продакшена — Redis (Этап 13).
    """

    def __init__(self, client: AsyncOpenAI | None = None):
        # client подменяется в тестах (dependency injection)
        self.client = client or AsyncOpenAI(
            api_key=settings.openai_api_key,
            timeout=settings.openai_timeout,
        )
        self.histories: Dict[str, List[dict]] = defaultdict(list)

    def _get_history(self, user_id: str) -> List[dict]:
        history = self.histories[user_id]
        if not history:
            history.append({"role": "system", "content": settings.system_prompt})
        return history

    def _trim_history(self, history: List[dict]) -> List[dict]:
        """Системный промпт (первый) + последние N сообщений."""
        if len(history) <= settings.max_history_length + 1:
            return history
        return [history[0]] + history[-settings.max_history_length:]

    async def handle_message(self, request: ChatRequest) -> ChatResponse:
        history = self._get_history(request.user_id)
        history.append({"role": "user", "content": request.message})
        history = self._trim_history(history)
        self.histories[request.user_id] = history

        logger.info(f"user={request.user_id[:8]} → OpenAI, history={len(history)} msgs")

        try:
            response = await self.client.chat.completions.create(
                model=settings.model,
                messages=history,
                temperature=0.8,
                max_tokens=1500,
                presence_penalty=0.2,
            )
        except OpenAIRateLimit:
            self._rollback(history)
            logger.warning("OpenAI rate limit hit")
            raise RateLimitError("Превышен лимит OpenAI. Попробуйте через минуту.")
        except APIError as e:
            self._rollback(history)
            logger.error(f"OpenAI APIError: {e}")
            raise OpenAIError("Ошибка AI-сервиса. Попробуйте позже.")
        except Exception as e:
            self._rollback(history)
            logger.exception("Unexpected error")
            raise OpenAIError(f"Неожиданная ошибка: {e}")

        reply = response.choices[0].message.content or ""
        history.append({"role": "assistant", "content": reply})
        self.histories[request.user_id] = history

        return ChatResponse(
            reply=reply,
            timestamp=datetime.now(timezone.utc).isoformat(),
        )

    @staticmethod
    def _rollback(history: List[dict]) -> None:
        """Убираем сообщение пользователя, на которое не получен ответ."""
        if history and history[-1]["role"] == "user":
            history.pop()

    def clear_history(self, user_id: str) -> None:
        self.histories.pop(user_id, None)
        logger.info(f"history cleared for user={user_id[:8]}")


chat_service = ChatService()

2.9. app/api/v1/chat.py

from fastapi import APIRouter, HTTPException, status
from app.models import ChatRequest, ChatResponse
from app.services.chat_service import chat_service
from app.services.rate_limiter import rate_limiter
from app.exceptions import OpenAIError, RateLimitError
from app.logger import logger

router = APIRouter(prefix="/chat", tags=["Chat"])


@router.post("", response_model=ChatResponse)
async def chat_endpoint(request: ChatRequest):
    """Основной endpoint общения с AI-мастером."""
    try:
        rate_limiter.check(request.user_id)
        return await chat_service.handle_message(request)
    except RateLimitError as e:
        raise HTTPException(status.HTTP_429_TOO_MANY_REQUESTS, detail=str(e))
    except OpenAIError as e:
        raise HTTPException(status.HTTP_502_BAD_GATEWAY, detail=str(e))
    except HTTPException:
        raise
    except Exception:
        logger.exception("Unhandled error in /chat")
        raise HTTPException(status.HTTP_500_INTERNAL_SERVER_ERROR,
                            detail="Внутренняя ошибка сервера")


@router.delete("/history/{user_id}")
async def clear_history(user_id: str):
    """Очистить историю диалога на сервере."""
    chat_service.clear_history(user_id)
    return {"message": "История очищена"}

2.10. app/main.py (исправлен CORS)

from contextlib import asynccontextmanager
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from app.config import settings
from app.api.v1.chat import router as chat_router
from app.models import HealthResponse
from app.logger import logger


@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info(f"🚀 Запуск {settings.app_name}, модель: {settings.model}")
    yield
    logger.info("🛑 Остановка сервера")


app = FastAPI(
    title=settings.app_name,
    description="RPG-чат с AI-мастером для D&D",
    version="1.0.0",
    lifespan=lifespan,
    docs_url="/docs" if settings.debug else None,
    redoc_url="/redoc" if settings.debug else None,
)

# ⚠️ allow_credentials=False: комбинация True + origins ["*"] запрещена спецификацией CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.cors_origins,
    allow_credentials=False,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(chat_router, prefix="/api/v1")


@app.get("/health", response_model=HealthResponse)
async def health_check():
    return HealthResponse(status="healthy")


@app.get("/")
async def root():
    return {"app": settings.app_name, "version": "1.0.0", "status": "running"}
